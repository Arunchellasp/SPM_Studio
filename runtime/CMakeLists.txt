cmake_minimum_required(VERSION 3.20)
project(spm_rt VERSION 0.3.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# third-party code is compiled as it is (no -Werror on code we do not own)
add_library(cjson STATIC third_party/cjson/cJSON.c)
target_include_directories(cjson PUBLIC third_party/cjson)
# step 36: civetweb, the WebSocket server (no TLS, no CGI, never serves files)
add_library(civetweb STATIC third_party/civetweb/civetweb.c)
target_include_directories(civetweb PUBLIC third_party/civetweb)
target_compile_definitions(civetweb PUBLIC NO_SSL NO_CGI NO_CACHING USE_WEBSOCKET NO_FILES)
if(MINGW)
  target_link_libraries(civetweb PUBLIC ws2_32 pthread) # MinGW: clock_gettime is in winpthreads (linked static)
elseif(WIN32)
  target_link_libraries(civetweb PUBLIC ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(civetweb PUBLIC Threads::Threads)
endif()

# our code: warnings are errors (design: -Wall -Werror)
function(spm_strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /wd4996)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wshadow -Werror)
  endif()
endfunction()

function(spm_libs target)
  target_link_libraries(${target} PRIVATE cjson civetweb)
  if(NOT WIN32)
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE) # before any header: CPU affinity, clock_nanosleep
  endif()
  if(WIN32)
    target_link_libraries(${target} PRIVATE winmm)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(${target} PRIVATE Threads::Threads m)
  endif()
endfunction()

# Unity build: spm_rt_main.c includes every layer (src/spm_rt_all.c, Software implementation › 11c)
add_executable(spm_rt src/spm_rt_main.c)
target_include_directories(spm_rt PRIVATE include src)
target_compile_definitions(spm_rt PRIVATE SPM_RT_VERSION="${PROJECT_VERSION}")
spm_strict(spm_rt)
spm_libs(spm_rt)
if(MINGW)
  target_link_options(spm_rt PRIVATE -static) # one spm_rt.exe, no MinGW DLLs next to it
endif()

# Unity tests (C): ctest runs them
enable_testing()
add_library(unity STATIC third_party/unity/unity.c)
target_include_directories(unity PUBLIC third_party/unity)
target_compile_definitions(unity PUBLIC UNITY_INCLUDE_DOUBLE) # TEST_ASSERT_DOUBLE_WITHIN
add_executable(spm_rt_tests tests/test_runtime.c)
target_include_directories(spm_rt_tests PRIVATE include src)
target_compile_definitions(spm_rt_tests PRIVATE SPM_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests" SPM_PROJECTS_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../projects")
spm_strict(spm_rt_tests)
spm_libs(spm_rt_tests)
target_link_libraries(spm_rt_tests PRIVATE unity)
if(MINGW)
  target_link_options(spm_rt_tests PRIVATE -static)
endif()
add_test(NAME runtime_unit COMMAND spm_rt_tests)
